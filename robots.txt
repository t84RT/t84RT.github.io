# ----------------------------------------------------------------------------
#  🔍 第三梯队：通用搜索引擎与全局规则（放在最后，作为兜底）
#  此规则适用于所有未明确列出的User-agent，包括常规搜索引擎和未知爬虫。
#  我们允许它们抓取，但通过Disallow限制一些无意义的路径。
# ----------------------------------------------------------------------------
User-agent: *
Allow: /

# 限制带查询参数的URL (如 ?page=2&sort=desc) ，避免产生无限多的重复页面。
# 这些动态URL通常不被搜索引擎青睐，且会浪费抓取配额。
Disallow: /*?*

# 禁止抓取临时文件和备份目录
Disallow: /assets/private/
Disallow: /tmp/
Disallow: /.git/
Disallow: /backup/
Disallow: /temp/

# 可选：限制通用爬虫抓取速度（单位：秒，非强制，但被部分爬虫遵守）
# 设置 Crawl-delay 为 1 秒，可有效降低服务器瞬时压力，尤其对GitHub Pages免费托管非常友好。
Crawl-delay: 1
